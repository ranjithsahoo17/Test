package com.eurocopter.mgs.download.analysis.health.impl.filtering;

import java.util.List;
import java.util.ListIterator;
import java.util.Set;

import com.eurocopter.mgs.datamodel.health.monitoring.ContextualParameter;
import com.eurocopter.mgs.datamodel.health.monitoring.FilteringLaw;
import com.eurocopter.mgs.download.analysis.health.data.IHealthIndicatorValueAcquisition;
import com.eurocopter.mgs.download.analysis.health.impl.HealthIndicatorValueAdapter;

public class HelionixFilteringPerFileAlgorithm extends AbstractFilteringRuleAlgorithm<String> {

   @Override
   protected boolean isFilteringLawApplicable(List<HealthIndicatorValueAdapter> indicatorValueAdapters) {
      return true;
   }

   @Override
   protected String getKey(IHealthIndicatorValueAcquisition acquisition) {
      return acquisition.getFileId();
   }

   @Override
   protected int getMaxPointsAtKey(String key, int configuredMaxPointsPerHour) {
      // Per file, the number of point is the configured one
      return configuredMaxPointsPerHour;
   }

   @Override
   protected void applyFilteringLawAlgorithm(List<HealthIndicatorValueAdapter> values, Set<FilteringLaw> filteringLaws, String indicatorFilteringLaw) {
      for (FilteringLaw filteringLaw : filteringLaws) {
         if (!filteringLaw.getName().equals(indicatorFilteringLaw)) {
            continue;
         }
         final Set<ContextualParameter> contextualParameters = filteringLaw.getContextualParameters();
         for (ContextualParameter contextualParameter : contextualParameters) {
            String name = contextualParameter.getName();
            double min = contextualParameter.getMinValue();
            double max = contextualParameter.getMaxValue();
            double s = contextualParameter.getThresholdOfVariabilityValue();

            ListIterator<HealthIndicatorValueAdapter> iter = values.listIterator();

            while (iter.hasNext()) {
               HealthIndicatorValueAdapter healthIndicatorValue = iter.next();
               if (healthIndicatorValue.getMinContextualValues().get(name) != null) {
                  double minValueContextualParameter = healthIndicatorValue.getMinContextualValues().get(name).getValue();
                  double maxValueContextualParameter = healthIndicatorValue.getMaxContextualValues().get(name).getValue();
                  double meanValueContextualParameter = healthIndicatorValue.getMeanContextualValues().get(name).getValue();

                  // Use the Filtering rules
                  if (!(min <= meanValueContextualParameter) || !(meanValueContextualParameter <= max)) {
                     iter.remove();
                  } else {
                     if (!((maxValueContextualParameter - minValueContextualParameter) <= s)) {
                        iter.remove();
                     }
                  }
               } else {
                  // The contextual parameter is not present for this indicator value
                  // We have to remove this indicator value as well
                  iter.remove();
               }
            }
         }
      }
   }
}
