

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

import org.eclipse.jface.viewers.ArrayContentProvider;
import org.eclipse.jface.viewers.CellEditor;
import org.eclipse.jface.viewers.CheckboxCellEditor;
import org.eclipse.jface.viewers.ComboBoxViewerCellEditor;
import org.eclipse.jface.viewers.ICellEditorListener;
import org.eclipse.jface.viewers.LabelProvider;
import org.eclipse.jface.viewers.TextCellEditor;
import org.eclipse.jface.viewers.TreeViewer;
import org.eclipse.swt.SWT;
import org.eclipse.swt.widgets.Text;
import org.eclipse.swt.widgets.Tree;
import org.eclipse.swt.widgets.TreeColumn;

import com.eurocopter.mgs.core.presentation.i18n.Messages;
import com.eurocopter.mgs.datamodel.aircraft.Aircraft;
import com.eurocopter.mgs.datamodel.health.monitoring.ThresholdLevel;
import com.eurocopter.mgs.datamodel.health.monitoring.ThresholdType;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.WCMechanicalComponentModel;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.WCModelDisplayProperty;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.store.HealthAlarmsPreferenceStore;

public class WCModelTreeViewer extends TreeViewer {

   protected Map<Aircraft, WCMechanicalComponentModel> healthWcModelMap;
   protected HealthAlarmsPreferenceStore healthAlarmsPreferenceStore;

   private Aircraft selectedAircraft = null;

   public WCModelTreeViewer(Tree tree, Map<Aircraft, WCMechanicalComponentModel> healthWcModelMap, HealthAlarmsPreferenceStore healthAlarmsPreferenceStore) {
      super(tree);
      this.healthWcModelMap = healthWcModelMap;
      this.healthAlarmsPreferenceStore = healthAlarmsPreferenceStore;
      createColumns(tree);

      String[] properties = new String[WCModelDisplayProperty.values().length];
      int i = 0;
      for (WCModelDisplayProperty prop : WCModelDisplayProperty.values()) {
         properties[i++] = prop.getKey();
      }
      setColumnProperties(properties);
   }

   /**
    * Create columns of the tree.
    * 
    * @param tree
    *           the tree for which columns have to be created
    * @return the list of columns name
    */
   private void createColumns(Tree tree) {

      final WCModelDisplayProperty[] properties = WCModelDisplayProperty.values();

      for (WCModelDisplayProperty property : properties) {
         final TreeColumn column = new TreeColumn(tree, SWT.CENTER);
         column.setText(property.getKey().isEmpty() ? "" : Messages.CORE.getMnemonicEscapedString(property.getKey()));
         column.setWidth(property.getColumnSize());
      }

      createCellEditors(tree, properties);
   }

   private void createCellEditors(Tree tree, final WCModelDisplayProperty[] propertyNames) {
      CellEditor[] editors = new CellEditor[propertyNames.length];
      /* editors[WCModelDisplayProperty.CHECKBOX.ordinal()] = getCellEditor(tree, false); */ // BCR-6605 Check Box Remove from cell editor.
      ComboBoxViewerCellEditor cellEditor = new ComboBoxViewerCellEditor(tree, SWT.READ_ONLY);
      cellEditor.setLabelProvider(new LabelProvider() {
         @Override
         public String getText(Object element) {
            if (element instanceof ThresholdLevel level) {
               return Messages.CORE.getString("ThresholdLevel_" + level.name());
            }
            return null;
         }
      });
      cellEditor.addListener(createWCColumnListener());
      cellEditor.setContentProvider(new ArrayContentProvider());
      List<ThresholdLevel> levels = new ArrayList<>();
      levels.add(ThresholdLevel.LOW);
      levels.add(ThresholdLevel.HIGH);
      cellEditor.setInput(levels);
      editors[WCModelDisplayProperty.LEVEL.ordinal()] = cellEditor;

      cellEditor = new ComboBoxViewerCellEditor(tree, SWT.READ_ONLY);
      cellEditor.setLabelProvider(new LabelProvider() {
         @Override
         public String getText(Object element) {
            if (element instanceof ThresholdType type) {
               return Messages.CORE.getString("ThresholdType_" + type.name());
            }
            return null;
         }
      });
      cellEditor.addListener(createWCColumnListener());
      cellEditor.setContentProvider(new ArrayContentProvider());
      List<ThresholdType> types = new ArrayList<>();
      types.add(ThresholdType.FIXED);
      types.add(ThresholdType.LEARNED);
      cellEditor.setInput(types);

      editors[WCModelDisplayProperty.TYPE.ordinal()] = cellEditor;

      editors[WCModelDisplayProperty.VALUE.ordinal()] = getCellEditor(tree, true);
      editors[WCModelDisplayProperty.MIN_VALUE.ordinal()] = getCellEditor(tree, true);
      editors[WCModelDisplayProperty.MAX_VALUE.ordinal()] = getCellEditor(tree, true);
      editors[WCModelDisplayProperty.LEARN_TIME.ordinal()] = getCellEditor(tree, true);

      TextCellEditor workcardEditor = (TextCellEditor) getCellEditor(tree, true);
      ((Text) workcardEditor.getControl()).setTextLimit(255);
      editors[WCModelDisplayProperty.WORKCARD_REF.ordinal()] = workcardEditor;

      editors[WCModelDisplayProperty.MIN_DEV.ordinal()] = getCellEditor(tree, true);
      editors[WCModelDisplayProperty.MEAN_MULT_FACTOR.ordinal()] = getCellEditor(tree, true);
      editors[WCModelDisplayProperty.DEV_COEF.ordinal()] = getCellEditor(tree, true);
      editors[WCModelDisplayProperty.NB_POINTS.ordinal()] = getCellEditor(tree, true);

      editors[WCModelDisplayProperty.MAX_TIME_WO_ACQ.ordinal()] = getCellEditor(tree, true);

      editors[WCModelDisplayProperty.FILTERING_LAW.ordinal()] = getCellEditor(tree, true); // added
      editors[WCModelDisplayProperty.TE.ordinal()] = getCellEditor(tree, true); // field edit

      super.setCellEditors(editors);
   }

   /**
    * 
    * @return
    */
   private CellEditor getCellEditor(Tree tree, boolean isTextEditor) {
      CellEditor cellEditor = isTextEditor ? new TextCellEditor(tree) : new CheckboxCellEditor(tree);
      cellEditor.addListener(createWCColumnListener());
      return cellEditor;
   }

   /**
    * @return
    */
   private ICellEditorListener createWCColumnListener() {
      return new ICellEditorListener() {
         @Override
         public void editorValueChanged(boolean arg0, boolean arg1) {
            // nothing to do
         }

         @Override
         public void cancelEditor() {
            // nothing to do
         }

         @Override
         public void applyEditorValue() {
            if (!healthWcModelMap.containsKey(selectedAircraft)) {
               healthWcModelMap.put(selectedAircraft, healthAlarmsPreferenceStore.getHealthModels(selectedAircraft));
            }
         }
      };
   }

   /**
    * @param selectedAircraft
    *           the selectedAircraft to set
    */
   public void setSelectedAircraft(Aircraft selectedAircraft) {
      this.selectedAircraft = selectedAircraft;
   }
}
