   /**
    * Gets a image from the image registry.
    *
    * @param key
    *           the image key
    * @return a image from the image registry that match the given key.
    */
   public static Image getImage(String key) {
      return Activator.getDefault().getImageRegistry().get(key);
   }

   /**
    * Gets the font.
    *
    * @param symbolicName
    *           The font key
    * @return the font;
    */
   public static Font getFont(String symbolicName) {
      return CoreResource.getFontRegistry().get(symbolicName);
   }

   /**
    * Gets the color.
    *
    * @param symbolicName
    *           color key.
    * @return the color.
    */
   public static Color getColor(String symbolicName) {
