   private static final String INVALID_CONF_FOR_AC_TYPE_MSG = "Health Monitoring Configuration file is not valid for the selected aircraft type";

   private static final HealthEvmDelegate EVM_DELEGATE = new HealthEvmDelegate();
   private static final HealthMechanicalComponentDelegate MECHANICAL_COMPONENT_DELEGATE = new HealthMechanicalComponentDelegate();
   private static final HealthThresholdDelegate THRESHOLD_DELEGATE = new HealthThresholdDelegate();

   public void createHealthSignalAnalysis(Aircraft aircraft, HealthConfigurationXMLBean healthConfigurationBean) throws DelegateException {
      // Check health configuration validity.
      if (!aircraft.getAircraftType().getName().equals(healthConfigurationBean.getAircraftType())) {
         throw new DelegateException(INVALID_CONF_FOR_AC_TYPE_MSG);
      }

      for (AccelerometerXMLBean accelerometerBean : healthConfigurationBean.getAccelerometers()) {
         Accelerometer accelerometer = new Accelerometer(UUID.randomUUID().toString());
         accelerometer.setFunctionalId(Accelerometer.normalize(accelerometerBean.getFunctionalId()));
         accelerometer.setName(accelerometerBean.getName());
         accelerometer.setActive(true);
         aircraft.getAccelerometers().add(accelerometer);
      }
   }

   public void createHealthMonitoring(Aircraft aircraft, HealthConfigurationXMLBean healthConfigurationBean, Map<String, byte[]> images, Map<String, Map<String, Unit>> unitsByPhysicalQuantity)
         throws DelegateException {
      // Check health configuration validity.
      if (!aircraft.getAircraftType().getName().equals(healthConfigurationBean.getAircraftType())) {
         throw new DelegateException(INVALID_CONF_FOR_AC_TYPE_MSG);
      }

      AircraftDisplayData acDisplayData = new AircraftDisplayData(UUID.randomUUID().toString());
      byte[] aircraftImage = images.get(healthConfigurationBean.getAircraftImg());
      if (aircraftImage == null) {
         throw new DelegateException("The A/C image cannot be found (" + healthConfigurationBean.getAircraftImg() + ")");
      }
      acDisplayData.setAcImage(aircraftImage);
      aircraft.setDisplayData(acDisplayData);

      final Avionic avionic = aircraft.getAircraftType().getAvionic();

      String currentPath = "";
      Set<HealthMechanicalComponent> acHealthComponents = aircraft.getHealthComponents();
      // Create mechanical components.
      for (MechanicalComponentXMLBean componentBean : healthConfigurationBean.getMechanicalComponents()) {
         acHealthComponents
               .addAll(MECHANICAL_COMPONENT_DELEGATE.createHealthMechanicalComponents(avionic, componentBean, acDisplayData, images, currentPath, unitsByPhysicalQuantity, healthConfigurationBean));
      }
   }

   public void updateHealthMonitoring(Aircraft aircraft, HealthConfigurationXMLBean healthConfigurationBean, Map<String, byte[]> images, Map<String, Map<String, Unit>> units)
         throws DelegateException {
      if (!aircraft.getAircraftType().getName().equals(healthConfigurationBean.getAircraftType())) {
         throw new DelegateException(INVALID_CONF_FOR_AC_TYPE_MSG);
      }

      // Index accelerometers by their functional ID
      Map<String, Accelerometer> accelerometers = new HashMap<>(aircraft.getAccelerometers().size());
      for (Accelerometer accelerometer : aircraft.getAccelerometers()) {
         accelerometers.put(accelerometer.getFunctionalId(), accelerometer);
         // Disable all accelerometers
         // --> accelerometers present in the new configuration will be enabled below
         accelerometer.setActive(false);
      }

      // Create or update accelerometers
      for (AccelerometerXMLBean accelerometerBean : healthConfigurationBean.getAccelerometers()) {
         updateAccelerometer(aircraft, accelerometers, accelerometerBean);
      }

      // Index components by their functional ID
      Map<String, HealthMechanicalComponent> hmcFuncIdHmcMap = new HashMap<>(aircraft.getHealthComponents().size());
      for (HealthMechanicalComponent healthComponent : aircraft.getHealthComponents()) {
         hmcFuncIdHmcMap.put(healthComponent.getFunctionalId(), healthComponent);
         // Disable all components
         // --> active components will be enabled by the following instructions
         healthComponent.setActive(false);
      }

      String rootPath = "";
      for (MechanicalComponentXMLBean componentBean : healthConfigurationBean.getMechanicalComponents()) {
         MECHANICAL_COMPONENT_DELEGATE.updateMechanicalComponent(aircraft, componentBean, hmcFuncIdHmcMap, accelerometers, rootPath, units, healthConfigurationBean);
      }

      // update AircraftDisplayData
      AircraftDisplayData acDisplayData = new AircraftDisplayData(UUID.randomUUID().toString());
      byte[] aircraftImage = images.get(healthConfigurationBean.getAircraftImg());
      if (aircraftImage == null) {
         throw new DelegateException("The A/C image cannot be found (" + healthConfigurationBean.getAircraftImg() + ")");
      }
      acDisplayData.setAcImage(aircraftImage);
      aircraft.setDisplayData(acDisplayData);
      aircraft.getDisplayData().setMechanicalComponentDisplayData(updateAircraftDisplayData(aircraft.getDisplayData(), healthConfigurationBean.getMechanicalComponents(), images));
   }

   public void createHealthMonitoringConfiguration(AircraftConfiguration aircraftConfiguration, HealthConfigurationXMLBean healthConfigurationBean,
         Map<Threshold, Set<RelearnThresholdRequest>> relearnRequests)
         throws DelegateException {
      Aircraft aircraft = aircraftConfiguration.getAircraft();
      // Check health configuration validity.
      if (!aircraft.getAircraftType().getName().equals(healthConfigurationBean.getAircraftType())) {
         throw new DelegateException(INVALID_CONF_FOR_AC_TYPE_MSG);
      }

      Set<FilteringLaw> filteringLaws = createFilteringLawConfiguration(aircraft.getAircraftType().getAvionic(), healthConfigurationBean);

      // Create health monitoring configuration
      HealthMonitoringConfiguration configuration = new HealthMonitoringConfiguration(UUID.randomUUID().toString());
      configuration.setVersion(healthConfigurationBean.getVersion());
      configuration.setOriginal(true);
      configuration.setFilteringLaws(filteringLaws);

      if (healthConfigurationBean.getVpuFailures() != null) {
         validateVpuComponentConfiguration(healthConfigurationBean.getVpuFailures());
         configuration.setVpuCounterLabels(createVpuCounterLabelConfiguration(healthConfigurationBean.getVpuFailures()));
         configuration.setVpuComponents(createVpuComponentConfiguration(configuration, healthConfigurationBean.getVpuFailures()));
         configuration.setVpuCounterWorkcard(healthConfigurationBean.getVpuFailures().getWorkcard());
      }

      aircraftConfiguration.getConfigurations().put(configuration.getConfigurationType(), configuration);

      // Index components by their functional ID
      Map<String, HealthMechanicalComponent> components = new HashMap<>(aircraft.getHealthComponents().size());

      for (HealthMechanicalComponent healthComponent : aircraft.getHealthComponents()) {
         components.put(healthComponent.getFunctionalId(), healthComponent);
      }

      // Create accelerometers.
      Map<String, Accelerometer> accelerometers = new HashMap<>(aircraft.getAccelerometers().size());
      for (Accelerometer accelerometer : aircraft.getAccelerometers()) {
         accelerometers.put(accelerometer.getFunctionalId(), accelerometer);
      }

      Avionic avionic = aircraft.getAircraftType().getAvionic();
      for (MechanicalComponentXMLBean componentBean : healthConfigurationBean.getMechanicalComponents()) {
         MECHANICAL_COMPONENT_DELEGATE.createMechanicalComponentConfiguration(avionic, configuration, componentBean, components, healthConfigurationBean.getEvmParameters(), accelerometers,
               healthConfigurationBean, relearnRequests);
      }

      if (EVM_DELEGATE.containsEvmComponents(configuration.getMechanicalComponentConfigurations())) {
         EvmParametersXMLBean evmParameters = healthConfigurationBean.getEvmParameters();
         EvmConfiguration evmConfiguration = new EvmConfiguration(UUID.randomUUID().toString());
         for (EvmAccelerometerXMLBean accelerometerBean : evmParameters.getAccelerometers()) {
            evmConfiguration.getAccelerometerConfigurations().add(EVM_DELEGATE.createEvmAccelerometerConfiguration(aircraft.getAircraftType().getAvionic(), accelerometerBean));
         }
         configuration.setEvmConfiguration(evmConfiguration);
      }
   }

   public Map<Threshold, Set<RelearnThresholdRequest>> getRelearnRequestsByThreshold(HealthMonitoringConfiguration healthMonitoringConfiguration) {
      Map<Threshold, Set<RelearnThresholdRequest>> requests = new HashMap<>();
      for (HealthMechanicalComponentConfiguration mechanicalComponentConfiguration : healthMonitoringConfiguration.getMechanicalComponentConfigurations().values()) {
         for (IndicatorConfiguration indicatorConfiguration : mechanicalComponentConfiguration.getIndicatorConfigurations().values()) {
            for (ThresholdConfiguration thresholdConfiguration : indicatorConfiguration.getThresholdConfigurations()) {
               if (thresholdConfiguration instanceof LearnedThresholdConfiguration learnedThresholdConfiguration) {
            	   requests.put(thresholdConfiguration.getThreshold(), learnedThresholdConfiguration.getRelearnRequests());
               }
            }
         }
      }
      return requests;
   }

   private Set<VpuComponent> createVpuComponentConfiguration(HealthMonitoringConfiguration healthMonitoringConfiguration, VpuFailuresXMLBean vpuFailures) throws DelegateException {
      Set<VpuComponent> components = new HashSet<>();
      for (VpuComponentXMLBean componentBean : vpuFailures.getComponents()) {
         VpuComponent component = new VpuComponent(UUID.randomUUID().toString());
         component.setBlock(componentBean.getBlockName().name());
         if (componentBean.getPhase() != null) {
            component.setPhase(componentBean.getPhase().name());
         }
         component.setEngineNumber(componentBean.getEngineNumber());
         component.setLabel(componentBean.getLabel());
         component.setVpuFailureCounters(MECHANICAL_COMPONENT_DELEGATE.processVpuFailureCounters(healthMonitoringConfiguration, componentBean.getFailureCounters(), component.getBlock()));
         components.add(component);
      }
      return components;
   }

   private Set<VpuCounterLabel> createVpuCounterLabelConfiguration(VpuFailuresXMLBean vpuFailuresXMLBean) throws DelegateException {
      Map<VpuCounterName, VpuCounterLabel> vpuCounterLabels = new EnumMap<>(VpuCounterName.class);
      for (VpuFailureCounterXMLBean failureCounterBean : vpuFailuresXMLBean.getFailureCounters()) {
         VpuCounterName vpuCounterName = VpuCounterName.valueOf(failureCounterBean.getId().name());
         if (vpuCounterLabels.containsKey(vpuCounterName)) {
            throw new DelegateException("Health Monitoring Configuration file is not valid: VPU Failure counter label for '" + failureCounterBean.getId() + "' is defined twice.");
         }
         VpuCounterLabel counterLabel = new VpuCounterLabel(UUID.randomUUID().toString());
         counterLabel.setName(vpuCounterName);
         counterLabel.setLabel(failureCounterBean.getLabel());
         vpuCounterLabels.put(vpuCounterName, counterLabel);
      }
      return new HashSet<>(vpuCounterLabels.values());
   }

   private void validateVpuComponentConfiguration(VpuFailuresXMLBean vpuFailuresXMLBean) throws DelegateException {
      // This set is only here to check uniqueness in configuration
      // I.e. a natural key is built for each component configured
      // And if the natural key is found twice, it raises an error
      Set<String> processedBlocks = new HashSet<>();
      for (VpuComponentXMLBean component : vpuFailuresXMLBean.getComponents()) {
         final String functionalKey;
         if (component.getBlockName() == VpuBlockName.SSVM) {
            if (component.getPhase() == VpuPhase.POWER_UP) {
               functionalKey = "SSVM Power-Up " + (component.getEngineNumber() == null ? "" : "Eng " + component.getEngineNumber());
            } else if (component.getPhase() == VpuPhase.STABILIZED) {
               if (component.getEngineNumber() == null) {
                  throw new DelegateException("VPU Failures Configuration: 'engine' attribute is mandatory for SSVM Stabilized block.");
               }
               functionalKey = "SSVM Stabilized Eng " + component.getEngineNumber();
            } else {
               throw new DelegateException("VPU Failures Configuration: 'phase' attribute is mandatory for SSVM block");
            }
         } else if (component.getBlockName() == VpuBlockName.SBRP || component.getBlockName() == VpuBlockName.SBRA) {
            if (component.getPhase() != null) {
               throw new DelegateException("VPU Failures Configuration: No 'phase' attribute shall be present for block " + component.getBlockName());
            }
            if (component.getEngineNumber() != null) {
               throw new DelegateException("VPU Failures Configuration: No 'engine' attribute shall be present for block " + component.getBlockName());
            }
            functionalKey = component.getBlockName().name();
         } else {
            throw new DelegateException("VPU Failures Configuration: Invalid block " + component.getBlockName());
         }
         if (processedBlocks.contains(functionalKey)) {
            throw new DelegateException("VPU Failures Configuration: Duplicated block " + functionalKey);
         }
         processedBlocks.add(functionalKey);
      }
   }

   private Set<FilteringLaw> createFilteringLawConfiguration(Avionic avionic, HealthConfigurationXMLBean healthConfigurationBean) throws DelegateException {
      Map<String, FilteringLaw> mapFilteringLaws = new HashMap<>();
      for (FilteringLawXMLBean filteringLaw : healthConfigurationBean.getFilteringLaws()) {
         Set<ContextualParameter> contextualParameters = new HashSet<>();
         Set<String> listNameCtx = new HashSet<>();
         FilteringLaw temp = new FilteringLaw(UUID.randomUUID().toString());
         temp.setName(filteringLaw.getName());
         for (ContextualParameterXMLBean xmlContextualParameter : filteringLaw.getContextualParameter()) {
            ContextualParameter contextualParameter = new ContextualParameter(UUID.randomUUID().toString());
            contextualParameter.setName(xmlContextualParameter.getName());
            contextualParameter.setMinValue(xmlContextualParameter.getMin());
            contextualParameter.setMaxValue(xmlContextualParameter.getMax());
            if (avionic == Avionic.CIGALHE) {
               // On HELIONIX, the parameter 's' is mandatory
               if (xmlContextualParameter.getS() == null) {
                  throw new DelegateException("HealthMonitoring: The parameter 's' must be configured on filtering law (parameter = " + contextualParameter.getName() + ").");
               }
               contextualParameter.setThresholdOfVariabilityValue(xmlContextualParameter.getS());
            }
            // On M'ARMS, the parameter is ignored
            if (listNameCtx.contains(contextualParameter.getName())) {
               throw new DelegateException("HealthMonitoring:  the name of the contextual parameter '" + contextualParameter.getName() +
                     "' must be unique for the law '" + temp.getName() + "' of the xml file .");
            } else {
               listNameCtx.add(contextualParameter.getName());
               contextualParameters.add(contextualParameter);
            }
         }
         temp.setContextualParameters(contextualParameters);
         if (mapFilteringLaws.containsKey(temp.getName())) {
            throw new DelegateException("HealthMonitoring:  the name of the law '" + temp.getName() + "' must be unique in the xml file .");
         } else {
            mapFilteringLaws.put(temp.getName(), temp);
         }
      }

      return new HashSet<>(mapFilteringLaws.values());
   }

   public void updateHealthConfiguration(HealthMonitoringConfiguration healthMonitoringConfiguration, Set<MechanicalComponentConfigurationDto> mechanicalComponentDtos) {
      healthMonitoringConfiguration.setOriginal(false);

      Map<String, IndicatorConfiguration> indicatorConfigurationsMap = new HashMap<>();
      Map<String, HealthMechanicalComponentConfiguration> healthMechanicalComponentConfigurationsMap = new HashMap<>();
      toConfigurationMaps(healthMonitoringConfiguration, healthMechanicalComponentConfigurationsMap, indicatorConfigurationsMap);

      Map<Threshold, Set<RelearnThresholdRequest>> relearnThresholdRequests = getRelearnRequestsByThreshold(healthMonitoringConfiguration);

      for (MechanicalComponentConfigurationDto mechanicalComponentDto : mechanicalComponentDtos) {
         HealthMechanicalComponentConfiguration hmcc = healthMechanicalComponentConfigurationsMap.get(mechanicalComponentDto.getMechanicalComponentId());
         hmcc.setWorkcard(mechanicalComponentDto.getWorkcardReference());
         hmcc.setMaximumTimeWithoutAcquistion(mechanicalComponentDto.getMaxTimeWOAcquisition());

         for (IndicatorConfigurationDto indicatorConfigurationDto : mechanicalComponentDto.getIndicatorConfigurationDtos()) {
            IndicatorConfiguration indicatorConfiguration = indicatorConfigurationsMap.get(indicatorConfigurationDto.getIndicatorId());
            indicatorConfiguration.getThresholdConfigurations().clear();

            Map<Severity, Threshold> thresholdsMap = THRESHOLD_DELEGATE.toThresholdMap(indicatorConfiguration.getIndicator().getThresholds());
            for (ThresholdConfigurationDto thresholdConfigurationDto : indicatorConfigurationDto.getThresholdConfigurationDtos()) {
               Threshold threshold = thresholdsMap.get(thresholdConfigurationDto.getThresholdSeverity());
               if (threshold == null) {
                  threshold = THRESHOLD_DELEGATE.createThreshold(thresholdConfigurationDto);
                  indicatorConfiguration.getIndicator().getThresholds().add(threshold);
               }

               ThresholdConfiguration thresholdConfiguration = THRESHOLD_DELEGATE.createThresholdConfiguration(thresholdConfigurationDto, relearnThresholdRequests.get(threshold));
               thresholdConfiguration.setWorkcard(thresholdConfigurationDto.getWorkcard());
               thresholdConfiguration.setThreshold(threshold);
               indicatorConfiguration.getThresholdConfigurations().add(thresholdConfiguration);
            }
         }
      }
   }

   private void updateAccelerometer(Aircraft aircraft, Map<String, Accelerometer> accelerometers, AccelerometerXMLBean accelerometerBean) {
      Accelerometer accelerometer = accelerometers.get(Accelerometer.normalize(accelerometerBean.getFunctionalId()));
      if (accelerometer == null) {
         accelerometer = new Accelerometer(UUID.randomUUID().toString());
         accelerometer.setFunctionalId(Accelerometer.normalize(accelerometerBean.getFunctionalId()));
         accelerometer.setName(accelerometerBean.getName());
         accelerometer.setActive(true);
         aircraft.getAccelerometers().add(accelerometer);

         // Register accelerometer since it may be referenced by a new indicator
         accelerometers.put(accelerometer.getFunctionalId(), accelerometer);
      } else {
         accelerometer.setName(accelerometerBean.getName());
      }
      accelerometer.setActive(true);
   }

   private Map<String, MechanicalComponentDisplayData> updateAircraftDisplayData(AircraftDisplayData acDisplayData, List<MechanicalComponentXMLBean> mechanicalComponentXMLBeans,
         Map<String, byte[]> images) throws DelegateException {
      // Get the old Display Data Map
      Map<String, MechanicalComponentDisplayData> oldDDMap = new HashMap<>(acDisplayData.getMechanicalComponentDisplayData());

      // Create the new Display Data Map
      Map<String, MechanicalComponentDisplayData> newDDMap = new HashMap<>();
      String currentPath = "";
      for (MechanicalComponentXMLBean mechanicalComponentXMLBean : mechanicalComponentXMLBeans) {
         createDisplayDataMap(mechanicalComponentXMLBean, currentPath, images, newDDMap);
      }

      Map<String, MechanicalComponentDisplayData> updatedDDMap = new HashMap<>();

      // Get removed display datas
      Set<String> deletedPaths = new HashSet<>(oldDDMap.keySet());
      deletedPaths.removeAll(newDDMap.keySet());

      // Add and update old datas to the updated map
      for (Entry<String, MechanicalComponentDisplayData> entry : oldDDMap.entrySet()) {
         if (!deletedPaths.contains(entry.getKey())) {
            // Update and add (or just add) the old mechanical component display data to the updated map
            MechanicalComponentDisplayData newMCDD = newDDMap.get(entry.getKey());
            MechanicalComponentDisplayData updatedMCDD = entry.getValue();
            updatedMCDD.setImage(newMCDD.getImage());
            updatedMCDD.setImageForParentComponent(newMCDD.getImageForParentComponent());
            updatedMCDD.setNotInParentComponent(newMCDD.isNotInParentComponent());
            updatedMCDD.setxPositionOnAc(newMCDD.getxPositionOnAc());
            updatedMCDD.setyPositionOnAc(newMCDD.getyPositionOnAc());
            updatedDDMap.put(entry.getKey(), updatedMCDD);
         }
      }

      // Get new display datas
      Set<String> addedPaths = new HashSet<>(newDDMap.keySet());
      addedPaths.removeAll(oldDDMap.keySet());

      // Add new datas to the updated map
      for (Entry<String, MechanicalComponentDisplayData> entry : newDDMap.entrySet()) {
         if (addedPaths.contains(entry.getKey())) {
            updatedDDMap.put(entry.getKey(), entry.getValue());
         }
      }

      return updatedDDMap;
   }

   private void createDisplayDataMap(MechanicalComponentXMLBean mechanicalComponentXMLBean, String currentPath, Map<String, byte[]> images, Map<String, MechanicalComponentDisplayData> newDDMap)
         throws DelegateException {
      if (mechanicalComponentXMLBean instanceof HealthMechanicalComponentXMLBean) {
         newDDMap.put(currentPath + HealthMechanicalComponent.HMC_PATH_SEPARATOR + mechanicalComponentXMLBean.getName(),
               MECHANICAL_COMPONENT_DELEGATE.createMechanicalComponentDisplayData(mechanicalComponentXMLBean, images));
      } else {
         String path = currentPath + HealthMechanicalComponent.HMC_PATH_SEPARATOR + mechanicalComponentXMLBean.getName();
         newDDMap.put(path, MECHANICAL_COMPONENT_DELEGATE.createMechanicalComponentDisplayData(mechanicalComponentXMLBean, images));
         for (MechanicalComponentXMLBean subComponentBean : mechanicalComponentXMLBean.getSubComponents()) {
            createDisplayDataMap(subComponentBean, path, images, newDDMap);
         }
      }
   }

   private void toConfigurationMaps(HealthMonitoringConfiguration healthMonitoringConfiguration,
         Map<String, HealthMechanicalComponentConfiguration> hmcConfigurations, Map<String, IndicatorConfiguration> indicatorConfigurations) {
      for (Map.Entry<HealthMechanicalComponent, HealthMechanicalComponentConfiguration> healthMechanicalComponentConfigurationEntry : healthMonitoringConfiguration
            .getMechanicalComponentConfigurations().entrySet()) {
         for (Map.Entry<Indicator, IndicatorConfiguration> indicatorConfigurationEntry : healthMechanicalComponentConfigurationEntry.getValue().getIndicatorConfigurations().entrySet()) {
            indicatorConfigurations.put(indicatorConfigurationEntry.getKey().getId(), indicatorConfigurationEntry.getValue());
         }
         hmcConfigurations.put(healthMechanicalComponentConfigurationEntry.getKey().getId(), healthMechanicalComponentConfigurationEntry.getValue());
      }
   }
