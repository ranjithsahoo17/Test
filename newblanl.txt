

import java.util.HashMap;
import java.util.Map;

import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;
import org.eclipse.jface.viewers.ISelectionChangedListener;
import org.eclipse.swt.SWT;
import org.eclipse.swt.layout.GridData;
import org.eclipse.swt.layout.GridLayout;
import org.eclipse.swt.widgets.Composite;
import org.eclipse.swt.widgets.Label;
import org.eclipse.swt.widgets.Tree;

import com.eurocopter.mgs.datamodel.aircraft.Aircraft;
import com.eurocopter.mgs.presentation.configuration.preferences.health.AircraftSelectionComposite;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.editor.WCTreeCellModifier;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.provider.WCTreeContentProvider;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.provider.WCTreeLabelProvider;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.store.HealthAlarmsPreferenceStore;
import com.eurocopter.mgs.presentation.configuration.preferences.healthalarms.viewer.WCModelTreeViewer;
import com.eurocopter.mgs.presentation.configuration.store.MgsPreferenceStoreComposer;
import com.eurocopter.mgs.presentation.configuration.store.MgsPreferenceStoreManager;

public class HealthAlarmsComposite extends Composite {

   private static final Logger LOGGER = LogManager.getLogger(HealthAlarmsComposite.class);

   /** Hash Map for the health mechanical component. */
   private final Map<Aircraft, WCMechanicalComponentModel> healthWcModelMap;

   /** TreeViewer for the thresholds. */
   private WCModelTreeViewer thresholdsTreeViewer;

   /**
    * ComboBox - A/C Type.
    */
   private final AircraftSelectionComposite aircraftCombo;

   /**
    * The Key.
    */
   private Aircraft selectedAircraft = null;

   /**
    * Store.
    */
   private final HealthAlarmsPreferenceStore healthAlarmsPreferenceStore;

   /**
    * Default constructor. Builds a HealthAlarmsComposite.
    * 
    * @param parentStore
    *           the parent store
    * @param parent
    *           the parent composite
    * @param style
    *           the style
    */
   public HealthAlarmsComposite(MgsPreferenceStoreComposer parentStore, Composite parent) {
      super(parent, SWT.NONE);
      GridLayout layout = new GridLayout();
      layout.marginHeight = 0;
      layout.marginWidth = 0;
      layout.verticalSpacing = 0;
      setLayout(layout);
      setLayoutData(new GridData(SWT.FILL, SWT.FILL, true, true, 1, 1));

      // add the store
      healthAlarmsPreferenceStore = new HealthAlarmsPreferenceStore(this);
      parentStore.addStore(healthAlarmsPreferenceStore);

      // instantiate the hash map
      healthWcModelMap = new HashMap<>();

      // A/C type composite
      aircraftCombo = new AircraftSelectionComposite(this);
      aircraftCombo.forceFocus();

      // set a space between composite
      new Label(this, SWT.NONE);

      // creation of the treeView
      createTreeViewer(this);

      aircraftCombo.getComboViewer().addSelectionChangedListener((ISelectionChangedListener) event -> {
         selectedAircraft = aircraftCombo.getSelectedAircraft();
         if (selectedAircraft != null) {
            healthAlarmsPreferenceStore.initMechanicalComponentData(selectedAircraft);
            WCMechanicalComponentModel aircraftComponentModel = healthAlarmsPreferenceStore.getHealthModels(selectedAircraft);
            refreshViewer(selectedAircraft, aircraftComponentModel);
         }

      });
   }

   /**
    * Create the thresholds Tree Viewer.
    * 
    * @param composite
    *           Parent Composite
    */
   private void createTreeViewer(Composite composite) {
      final Tree tree = new Tree(composite, SWT.BORDER | SWT.H_SCROLL | SWT.V_SCROLL | SWT.FULL_SELECTION);
      tree.setLayoutData(new GridData(SWT.FILL, SWT.FILL, true, true, 2, 1));
      tree.setHeaderVisible(true);
      tree.setLinesVisible(true);
      thresholdsTreeViewer = new WCModelTreeViewer(tree, healthWcModelMap, healthAlarmsPreferenceStore);
      thresholdsTreeViewer.setLabelProvider(new WCTreeLabelProvider());
      thresholdsTreeViewer.setContentProvider(new WCTreeContentProvider());
      thresholdsTreeViewer.setCellModifier(new WCTreeCellModifier(thresholdsTreeViewer));
   }

   /**
    * Refresh Viewer.
    * 
    * @param wcModel
    *           List<HealthWCModel> models.
    */
   private void refreshViewer(Aircraft aircraft, WCMechanicalComponentModel wcModel) {
      thresholdsTreeViewer.setData("aircraft", aircraft);
      thresholdsTreeViewer.setSelectedAircraft(aircraft);
      thresholdsTreeViewer.setInput(wcModel);
      thresholdsTreeViewer.expandAll();
      thresholdsTreeViewer.refresh();
   }

   /**
    * Perform Apply for the preference page.
    */
   public void performApply() {
      healthAlarmsPreferenceStore.innerSave();
      try {
         MgsPreferenceStoreManager.saveAircraftConfigurations();
         MgsPreferenceStoreManager.clearAircraftConfigurations();
      } catch (Exception e) {
         LOGGER.error("Error while saving health preferences", e);
      }
   }

   /**
    * Validate data before setting them.
    * 
    * @return true if data are valid, false otherwise
    */
   public boolean isValidData() {
      return healthAlarmsPreferenceStore.isValidData(healthWcModelMap);
   }

   /**
    * Perform Defaults for the preference page.
    */
   public void performDefaults() {
      if (selectedAircraft != null) {
         healthWcModelMap.clear();
         refreshViewer(selectedAircraft, healthAlarmsPreferenceStore.loadHealthModels(selectedAircraft));
      }
   }

   /**
    * Cancel was pressed.
    */
   public void performCancel() {
      // empty the store, otherwise data will be saved because of our handling of needSaving method.
      healthAlarmsPreferenceStore.emptyData();
   }
}
