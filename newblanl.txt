

public enum WCModelDisplayProperty {
   COMPONENT("MonitoredComponents", 200), CHECKBOX("", 20), SEVERITY("Severity", 60), LEVEL("ThresholdLevel", 60), TYPE("Type", 80), VALUE("ThresholdValue", 50), MIN_VALUE("LearningThresholdMinValue",
         50), MAX_VALUE("LearningThresholdMaxValue", 50), LEARN_TIME("healthAlarmPageColumn8", 50), WORKCARD_REF("WorkCardRef", 100), MIN_DEV("LearningThresholdMinDev", 50), MEAN_MULT_FACTOR(
               "LearningThresholdMeanMultFactor", 50), DEV_COEF("LearningThresholdDevCoef",
                     50), NB_POINTS("NbSuccessivePtForAlarmRaising", 50), MAX_TIME_WO_ACQ("MaxTimeWithoutAcquisition", 60), FILTERING_LAW("FilteringLaw", 100), TE("TeReference", 120);

   private final String key;

   private final int columnSize;

   WCModelDisplayProperty(String key, int columnSize) {
      this.key = key;
      this.columnSize = columnSize;
   }

   public String getKey() {
      return key;
   }

   public int getColumnSize() {
      return columnSize;
   }

   public static WCModelDisplayProperty fromKey(String property) {
      for (WCModelDisplayProperty prop : values()) {
         if (prop.getKey().equals(property)) {
            return prop;
         }
      }
      return null;
   }
}
