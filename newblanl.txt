

import java.util.Calendar;

import com.eurocopter.mgs.datamodel.health.monitoring.Indicator;

import com.eurocopter.mgs.datamodel.health.monitoring.RelearnThresholdRequest;
import com.eurocopter.mgs.datamodel.health.monitoring.Threshold;

import com.eurocopter.mgs.xml.aircraftconfiguration.health.common.ThresholdTypeXMLBean;
import com.eurocopter.mgs.xml.aircraftconfiguration.health.common.ThresholdXMLBean;

class HealthThresholdDelegate {

   private static final long ONE_MINUTE = 60L;

   void createThreshold(Indicator indicator, ThresholdXMLBean thresholdXMLBean) {
      final Threshold threshold = new Threshold(UUID.randomUUID().toString());
      threshold.setActive(true);
      threshold.setFunctionalId(thresholdXMLBean.getName());
      threshold.setSeverity(Severity.valueOf(thresholdXMLBean.getSeverity().name()));
      indicator.getThresholds().add(threshold);
   }

   void updateThreshold(Threshold threshold, ThresholdXMLBean thresholdXMLBean) {
      threshold.setActive(true);
      threshold.setSeverity(Severity.valueOf(thresholdXMLBean.getSeverity().name()));
   }

   void checkThresholdValues(IndicatorXMLBean indicatorXMLBean, String hmcFuncId) throws DelegateException {
      ThresholdXMLBean redModel = null;
      ThresholdXMLBean amberModel = null;

      for (ThresholdXMLBean currentThreshold : indicatorXMLBean.getThresholds()) {
         if (currentThreshold.getType() == ThresholdTypeXMLBean.FIXED && currentThreshold.getValue() == null) {
            throw new DelegateException("Issue with health configuration concerning " + hmcFuncId + " > "
                  + indicatorXMLBean.getFunctionalId() + " thresholds values: fixed threshold value is undefined.");
         }
         if (currentThreshold.getType() == ThresholdTypeXMLBean.LEARNED) {
            if (currentThreshold.getMaximumThreshold() == null) {
               throw new DelegateException("Issue with health configuration concerning " + hmcFuncId + " > "
                     + indicatorXMLBean.getFunctionalId() + " thresholds values: learned maximum value must be defined.");
            }
            if (currentThreshold.getMinimumThreshold() != null) {
               validateMinMaxThreshold(indicatorXMLBean, hmcFuncId, currentThreshold);
            }
         }
         if (currentThreshold.getSeverity() == ThresholdSeverityXMLBean.RED) {
            redModel = currentThreshold;
         } else {
            amberModel = currentThreshold;
         }

         if (redModel != null && amberModel != null) {

            boolean valid = true;

            if (amberModel.getValue() == null || redModel.getValue() == null) {
               // learned threshold without default value --> Nothing more to check
               // (fixed threshold without value cannot occur at this step --> checked before)
               continue;
            } else {
               boolean isAmberHigh = amberModel.getLevel() == ThresholdLevelXMLBean.HIGH;
               boolean isAmberLow = amberModel.getLevel() == ThresholdLevelXMLBean.LOW;
               boolean amberHigherThanRed = amberModel.getValue() > redModel.getValue();
               boolean amberLowerThanRed = amberModel.getValue() < redModel.getValue();
               if (amberModel.getLevel() == redModel.getLevel()) {
                  if (isAmberHigh && amberHigherThanRed || isAmberLow && amberLowerThanRed) {
                     // the value of the high amber threshold is greater than the value of the high red threshold
                     // or
                     // the value of the low amber threshold is smaller than the value of the low red threshold
                     valid = false;
                  }
               } else if (isAmberHigh && amberLowerThanRed || !isAmberHigh && amberHigherThanRed) {
                  // the value of the low threshold is greater than the value of the high threshold
                  // or
                  // the value of the low threshold is greater than the value of the high threshold
                  valid = false;
               }
            }
            if (!valid) {
               throw new DelegateException("Issue with health configuration concerning " + hmcFuncId + " > "
                     + indicatorXMLBean.getFunctionalId() + " thresholds values: check red and amber values.");
            }
         }
      }
   }

   void createThresholdConfiguration(IndicatorConfiguration indicatorConfiguration, ThresholdXMLBean thresholdBean, Map<String, Threshold> thresholds,
         Map<Threshold, Set<RelearnThresholdRequest>> relearnRequests) throws DelegateException {
      Threshold threshold = thresholds.get(thresholdBean.getName());
      Set<RelearnThresholdRequest> relearnThresholdRequests = relearnRequests.get(threshold);

      final ThresholdConfiguration thresholdConfiguration;
      if (thresholdBean.getType() == ThresholdTypeXMLBean.LEARNED) {
         LearnedThresholdConfiguration learnedThresholdConfiguration = new LearnedThresholdConfiguration(UUID.randomUUID().toString());
         learnedThresholdConfiguration.setThreshold(threshold);

         Duration learningTime;
         Float learningTimeBean = thresholdBean.getLearningTime();
         if (learningTimeBean != null) {
            learningTime = Duration.standardMinutes(learningTimeBean.longValue() * ONE_MINUTE);
         } else {
            throw new DelegateException("Learning time undefined");
         }

         learnedThresholdConfiguration.setLearningTime(learningTime.getMillis());         
         learnedThresholdConfiguration.setMinimumThreshold(thresholdBean.getMinimumThreshold());
         learnedThresholdConfiguration.setMaximumThreshold(thresholdBean.getMaximumThreshold());
         learnedThresholdConfiguration.setMinimumDeviation(thresholdBean.getMinimumDeviation());
         learnedThresholdConfiguration.setDeviationCoefficient(thresholdBean.getDeviationCoefficient());
         learnedThresholdConfiguration.setMeanFactor(thresholdBean.getMeanFactor());
         learnedThresholdConfiguration.setThresholdLevel(ThresholdLevel.valueOf(thresholdBean.getLevel().name()));
         learnedThresholdConfiguration.setValue(thresholdBean.getValue());
         
         thresholdConfiguration = learnedThresholdConfiguration;
         
      } else {
         FixedThresholdConfiguration fixedThresholdConfiguration = new FixedThresholdConfiguration(UUID.randomUUID().toString());
         fixedThresholdConfiguration.setThreshold(threshold);
         fixedThresholdConfiguration.setThresholdLevel(ThresholdLevel.valueOf(thresholdBean.getLevel().name()));
         fixedThresholdConfiguration.setValue(thresholdBean.getValue());
         thresholdConfiguration = fixedThresholdConfiguration;
      }

      thresholdConfiguration.setWorkcard(thresholdBean.getWorkcard());
      if (thresholdBean.getNbSuccessivePoints() != null) {
         int successivePoints = thresholdBean.getNbSuccessivePoints();
         if (successivePoints <= 0) {
            throw new DelegateException("The 'successive points' parameter must be positive for threshold " + thresholdBean.getName());
         }
         thresholdConfiguration.setNbSuccessivePoints(successivePoints);
      }
      thresholdConfiguration.setMinimumEngineOilTemperature(thresholdBean.getMinimumEngineOilTemperature());
      thresholdConfiguration.setThreshold(threshold);
      
      if (thresholdBean.getType() == ThresholdTypeXMLBean.LEARNED) {
	      // Create automatically a relearn request for all learned thresholds
	      createRelearnRequests(thresholdBean.getValue(), relearnThresholdRequests, (LearnedThresholdConfiguration) thresholdConfiguration);
      }

      indicatorConfiguration.getThresholdConfigurations().add(thresholdConfiguration);
   }

   Map<Severity, Threshold> toThresholdMap(Set<Threshold> thresholds) {
      Map<Severity, Threshold> thresholdsMap = new EnumMap<>(Severity.class);
      for (Threshold threshold : thresholds) {
         thresholdsMap.put(threshold.getSeverity(), threshold);
      }
      return thresholdsMap;
   }

   Threshold createThreshold(ThresholdConfigurationDto thresholdConfigurationDto) {
      Threshold threshold = new Threshold(UUID.randomUUID().toString());
      threshold.setSeverity(thresholdConfigurationDto.getThresholdSeverity());
      threshold.setFunctionalId(thresholdConfigurationDto.getThresholdSeverity().name().toLowerCase(Locale.US));
      threshold.setActive(true);
      return threshold;
   }

   ThresholdConfiguration createThresholdConfiguration(ThresholdConfigurationDto thresholdConfigurationDto, Set<RelearnThresholdRequest> relearnThresholdRequests) {
      ThresholdConfiguration thresholdConfiguration;
      if (thresholdConfigurationDto.getThresholdType() == ThresholdType.FIXED) {
         thresholdConfiguration = new FixedThresholdConfiguration(UUID.randomUUID().toString());
         thresholdConfiguration.setThresholdLevel(thresholdConfigurationDto.getThresholdLevel());
         ((FixedThresholdConfiguration) thresholdConfiguration).setValue(thresholdConfigurationDto.getValue());         
      } else {
         thresholdConfiguration = new LearnedThresholdConfiguration(UUID.randomUUID().toString());
         ((LearnedThresholdConfiguration) thresholdConfiguration).setLearningTime(thresholdConfigurationDto.getLearningTime());
         ((LearnedThresholdConfiguration) thresholdConfiguration).setMinimumThreshold(thresholdConfigurationDto.getMinimumThreshold());
         ((LearnedThresholdConfiguration) thresholdConfiguration).setMaximumThreshold(thresholdConfigurationDto.getMaximumThreshold());
         ((LearnedThresholdConfiguration) thresholdConfiguration).setMinimumDeviation(thresholdConfigurationDto.getMinimumDeviation());
         ((LearnedThresholdConfiguration) thresholdConfiguration).setDeviationCoefficient(thresholdConfigurationDto.getDeviationCoefficient());
         ((LearnedThresholdConfiguration) thresholdConfiguration).setMeanFactor(thresholdConfigurationDto.getMeanFactor());
         thresholdConfiguration.setThresholdLevel(thresholdConfigurationDto.getThresholdLevel());
         ((LearnedThresholdConfiguration) thresholdConfiguration).setValue(thresholdConfigurationDto.getDefaultValue());
         
      }
      thresholdConfiguration.setNbSuccessivePoints(thresholdConfigurationDto.getNbSuccessivePoints());
      thresholdConfiguration.setMinimumEngineOilTemperature(thresholdConfigurationDto.getMinimumEngineOilTemperature());      

      if (thresholdConfigurationDto.getThresholdType() == ThresholdType.LEARNED) {
	      // Create automatically a relearn request for all learned thresholds
	      createRelearnRequests(thresholdConfigurationDto.getDefaultValue(), relearnThresholdRequests, (LearnedThresholdConfiguration) thresholdConfiguration);
      }
      
      return thresholdConfiguration;
   }

   private void createRelearnRequests(Double defaultValue, Set<RelearnThresholdRequest> relearnThresholdRequests, LearnedThresholdConfiguration learnedThresholdConfiguration) {
   	LearnedThresholdAnalyzer learnedThresholdAnalyzer = new LearnedThresholdAnalyzer();   	
   	if (relearnThresholdRequests == null) {
   		RelearnThresholdRequest relearnRequest = createDefaultRelearnRequest(defaultValue);
		   learnedThresholdConfiguration.getRelearnRequests().add(relearnRequest);
	   } else {
		   for (RelearnThresholdRequest relearnRequest : relearnThresholdRequests) {
		   	// If learned threshold value is out of bounds after learning, ajust it to the min/max of the new configuration
		   	// The learned value is null as long as the relearn request has not been handled yet
		   	Double learnedValue = relearnRequest.getLearnedValue();
		   	if (relearnRequest.isHandled() && learnedValue != null) {
		   		double adjustedLearnedValue = learnedThresholdAnalyzer.ajustThresholdBounds(learnedThresholdConfiguration, learnedValue);
		   		if (Double.compare(adjustedLearnedValue, learnedValue) != 0) {
		   			// Adjust threshold value to min/max if out of bounds
		   			learnedThresholdConfiguration.getRelearnRequests().add(createAdjustedCopyOfRelearnRequest(relearnRequest, adjustedLearnedValue));
		   			continue;
		   		}
		   	}
		   	learnedThresholdConfiguration.getRelearnRequests().add(createCopyOfRelearnRequest(relearnRequest));
		   }
	  }
   }
   
   private RelearnThresholdRequest createDefaultRelearnRequest(Double value) {
      // Create automatically a relearn request for learned thresholds without previous relearn request
      RelearnThresholdRequest relearnRequest = new RelearnThresholdRequest(UUID.randomUUID().toString());
      relearnRequest.setDescription("A/C Configuration update");
      Calendar startDate = Calendar.getInstance(TimeZone.getTimeZone("GMT"));
      relearnRequest.setStartDate(startDate.getTime());
      relearnRequest.setStartFlightDate(new Date(0));
      relearnRequest.setHandled(false);
      relearnRequest.setLearnedValue(value);
      relearnRequest.setCreateMarker(false);
      return relearnRequest;
   }

   private RelearnThresholdRequest createCopyOfRelearnRequest(RelearnThresholdRequest latestRelearnRequest) {
      // Create automatically a relearn request based on the previous relearn request
      RelearnThresholdRequest relearnRequest = new RelearnThresholdRequest(UUID.randomUUID().toString());
      relearnRequest.setDescription(latestRelearnRequest.getDescription());
      relearnRequest.setStartDate(latestRelearnRequest.getStartDate());
      relearnRequest.setHandled(latestRelearnRequest.isHandled());
      relearnRequest.setLearnedValue(latestRelearnRequest.getLearnedValue());
      relearnRequest.setApplicableFlightDate(latestRelearnRequest.getApplicableFlightDate());
      relearnRequest.setStartFlightDate(latestRelearnRequest.getStartFlightDate());
      relearnRequest.setEndDate(latestRelearnRequest.getEndDate());
      relearnRequest.setHistoryLearningTime(latestRelearnRequest.getHistoryLearningTime());
      relearnRequest.setCreateMarker(latestRelearnRequest.isCreateMarker());
      return relearnRequest;
   }
   
   private RelearnThresholdRequest createAdjustedCopyOfRelearnRequest(RelearnThresholdRequest latestRelearnRequest, double adjustedLearnedValue) {
      // Create automatically a relearn request with adjusted value (when learned value is out of bounds) based on the previous relearn request
      RelearnThresholdRequest relearnRequest = createCopyOfRelearnRequest(latestRelearnRequest);
      relearnRequest.setLearnedValue(adjustedLearnedValue);
      return relearnRequest;
   }

   private void validateMinMaxThreshold(IndicatorXMLBean indicatorXMLBean, String hmcFuncId, ThresholdXMLBean currentThreshold) throws DelegateException {
      final double min;
      final double max;
      if (currentThreshold.getLevel() == ThresholdLevelXMLBean.HIGH) {
         min = currentThreshold.getMinimumThreshold();
         max = currentThreshold.getMaximumThreshold();
      } else {
         min = currentThreshold.getMaximumThreshold();
         max = currentThreshold.getMinimumThreshold();
      }
      if (min >= max) {
         throw new DelegateException("Issue with health configuration concerning " + hmcFuncId + " > "
               + indicatorXMLBean.getFunctionalId() + " thresholds values: check min and max values.");
      }
   }
}
