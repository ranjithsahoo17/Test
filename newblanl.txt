

import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

import com.eurocopter.mgs.datamodel.aircraft.configuration.Configuration;
import com.eurocopter.mgs.datamodel.health.monitoring.vpu.VpuComponent;
import com.eurocopter.mgs.datamodel.health.monitoring.vpu.VpuCounterLabel;

/**
 * The Class HealthMonitoringConfiguration.
 */
public class HealthMonitoringConfiguration extends Configuration {

   /** The Constant serialVersionUID. */
   private static final long serialVersionUID = 1L;

   /** The Constant TYPE. */
   public static final String TYPE = "HEALTH_MONITORING_CONFIGURATION";

   /** The health mechanical component configurations. */
   private Map<HealthMechanicalComponent, HealthMechanicalComponentConfiguration> mechanicalComponentConfigurations = new HashMap<>();

   private EvmConfiguration evmConfiguration;

   private Set<FilteringLaw> filteringLaws = new HashSet<>();

   private Set<VpuCounterLabel> vpuCounterLabels = new HashSet<>();

   private Set<VpuComponent> vpuComponents = new HashSet<>();

   private String vpuCounterWorkcard;

   /**
    * Instantiates a new health monitoring configuration.
    */
   HealthMonitoringConfiguration() {
   }

   /**
    * Instantiates a new health monitoring configuration.
    *
    * @param id
    *           the id
    */
   public HealthMonitoringConfiguration(String id) {
      super(id);
   }

   public Map<HealthMechanicalComponent, HealthMechanicalComponentConfiguration> getMechanicalComponentConfigurations() {
      return mechanicalComponentConfigurations;
   }

   public EvmConfiguration getEvmConfiguration() {
      return evmConfiguration;
   }

   public void setEvmConfiguration(EvmConfiguration evmConfiguration) {
      this.evmConfiguration = evmConfiguration;
   }

   @Override
   public String getConfigurationType() {
      return TYPE;
   }

   public Set<FilteringLaw> getFilteringLaws() {
      return filteringLaws;
   }

   public void setFilteringLaws(Set<FilteringLaw> filteringLaws) {
      this.filteringLaws = filteringLaws;
   }

   public Set<VpuCounterLabel> getVpuCounterLabels() {
      return vpuCounterLabels;
   }

   public void setVpuCounterLabels(Set<VpuCounterLabel> vpuCounterLabels) {
      this.vpuCounterLabels = vpuCounterLabels;
   }

   public String getVpuCounterWorkcard() {
      return vpuCounterWorkcard;
   }

   public void setVpuCounterWorkcard(String vpuCounterWorkcard) {
      this.vpuCounterWorkcard = vpuCounterWorkcard;
   }

   public Set<VpuComponent> getVpuComponents() {
      return vpuComponents;
   }

   public void setVpuComponents(Set<VpuComponent> vpuComponents) {
      this.vpuComponents = vpuComponents;
   }
}
