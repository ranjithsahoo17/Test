package com.eurocopter.mgs.download.analysis.health.impl.filtering;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Map.Entry;
import java.util.Set;

import com.eurocopter.mgs.datamodel.health.monitoring.FilteringLaw;
import com.eurocopter.mgs.datamodel.health.monitoring.HealthMonitoringConfiguration;
import com.eurocopter.mgs.datamodel.health.monitoring.Indicator;
import com.eurocopter.mgs.datamodel.health.monitoring.IndicatorConfiguration;
import com.eurocopter.mgs.download.analysis.health.algo.FilteringAlgorithm;
import com.eurocopter.mgs.download.analysis.health.data.IHealthIndicatorValueAcquisition;
import com.eurocopter.mgs.download.analysis.health.impl.HealthIndicatorValueAdapter;

public abstract class AbstractFilteringRuleAlgorithm<K> implements IFilteringRuleAlgorithm {

   private static final Comparator<HealthIndicatorValueAdapter> INDICATOR_VALUE_COMPARATOR = Comparator.comparing(HealthIndicatorValueAdapter::getAcquisitionDate);
   private final Map<K, Map<IndicatorConfiguration, List<HealthIndicatorValueAdapter>>> healthIndicatorValueAdapterByFileIdAndIndConf;

   public AbstractFilteringRuleAlgorithm() {
      this.healthIndicatorValueAdapterByFileIdAndIndConf = new HashMap<>();
   }

   protected abstract boolean isFilteringLawApplicable(List<HealthIndicatorValueAdapter> indicatorValueAdapters);

   protected abstract K getKey(IHealthIndicatorValueAcquisition acquisition);

   protected abstract int getMaxPointsAtKey(K key, int configuredMaxPointsPerHour);

   protected abstract void applyFilteringLawAlgorithm(List<HealthIndicatorValueAdapter> values, Set<FilteringLaw> filteringLaws, String indicatorFilteringLaw);

   @Override
   public void recordAcquisition(IHealthIndicatorValueAcquisition acquisition, String indicatorId, Indicator indicator, IndicatorConfiguration indicatorConfiguration) {
      K key = getKey(acquisition);
      final Map<IndicatorConfiguration, List<HealthIndicatorValueAdapter>> indicatorsInFile = healthIndicatorValueAdapterByFileIdAndIndConf.computeIfAbsent(key, k -> new HashMap<>());
      final List<HealthIndicatorValueAdapter> indicatorValues = indicatorsInFile.computeIfAbsent(indicatorConfiguration, k -> new ArrayList<>());
      indicatorValues.add(new HealthIndicatorValueAdapter(acquisition, indicatorConfiguration, indicatorId, indicator));
   }

   @Override
   public List<HealthIndicatorValueAdapter> getFilteredValues(HealthMonitoringConfiguration healthConfiguration) {
      for (Map<IndicatorConfiguration, List<HealthIndicatorValueAdapter>> indicators : healthIndicatorValueAdapterByFileIdAndIndConf.values()) {
         for (Entry<IndicatorConfiguration, List<HealthIndicatorValueAdapter>> entry2 : indicators.entrySet()) {
            final IndicatorConfiguration indicatorConfiguration = entry2.getKey();
            final Set<FilteringLaw> filteringLaws = healthConfiguration.getFilteringLaws();
            List<HealthIndicatorValueAdapter> indicatorValueAdapters = entry2.getValue();
            if (indicatorConfiguration.getFilteringLaw() != null && isFilteringLawApplicable(indicatorValueAdapters)) {
               applyFilteringLawAlgorithm(indicatorValueAdapters, filteringLaws, indicatorConfiguration.getFilteringLaw());
            }
         }
      }

      final List<HealthIndicatorValueAdapter> valuesToSave = new ArrayList<>();
      for (Map.Entry<K, Map<IndicatorConfiguration, List<HealthIndicatorValueAdapter>>> indicatorsEntry : healthIndicatorValueAdapterByFileIdAndIndConf.entrySet()) {
         K key = indicatorsEntry.getKey();
         Map<IndicatorConfiguration, List<HealthIndicatorValueAdapter>> indicators = indicatorsEntry.getValue();
         for (Entry<IndicatorConfiguration, List<HealthIndicatorValueAdapter>> indicatorEntry : indicators.entrySet()) {
            final IndicatorConfiguration conf = indicatorEntry.getKey();
            final List<HealthIndicatorValueAdapter> healthIndicatorValueAdapterValues = indicatorEntry.getValue();
            if (conf.getMaxPointsPerHour() == null) {
               valuesToSave.addAll(healthIndicatorValueAdapterValues);
            } else {
               healthIndicatorValueAdapterValues.sort(INDICATOR_VALUE_COMPARATOR);
               int maxPointsPerHour = getMaxPointsAtKey(key, conf.getMaxPointsPerHour());
               if (maxPointsPerHour > 0) {
                  valuesToSave.addAll(FilteringAlgorithm.filterHealthIndicatorValues(healthIndicatorValueAdapterValues, maxPointsPerHour));
               }
            }
         }
      }
      return valuesToSave;
   }

}
