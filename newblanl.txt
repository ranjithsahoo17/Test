
import java.util.ArrayList;
import java.util.List;

import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;
import org.eclipse.jface.viewers.ArrayContentProvider;
import org.eclipse.jface.viewers.ComboViewer;
import org.eclipse.jface.viewers.IStructuredSelection;
import org.eclipse.jface.viewers.LabelProvider;
import org.eclipse.swt.SWT;
import org.eclipse.swt.layout.FormAttachment;
import org.eclipse.swt.layout.FormData;
import org.eclipse.swt.layout.FormLayout;
import org.eclipse.swt.widgets.Combo;
import org.eclipse.swt.widgets.Composite;
import org.eclipse.swt.widgets.Display;
import org.eclipse.swt.widgets.Label;
import org.osgi.framework.BundleContext;
import org.osgi.framework.FrameworkUtil;

import com.eurocopter.mgs.business.aircraft.api.IAircraftBusiness;
import com.eurocopter.mgs.core.presentation.i18n.Messages;
import com.eurocopter.mgs.datamodel.aircraft.Aircraft;
import com.eurocopter.mgs.datamodel.aircraft.AircraftState;
import com.eurocopter.mgs.datamodel.aircraft.configuration.AircraftConfiguration;
import com.eurocopter.mgs.datamodel.health.monitoring.HealthMonitoringConfiguration;
import com.eurocopter.mgs.service.exception.EntityNotFoundException;

/**
 *
 * Composite for the aircraft Type.
 * <P>
 *
 * <P>
 *
 */
public class AircraftSelectionComposite extends Composite {

   private static final Logger LOGGER = LogManager.getLogger(AircraftSelectionComposite.class);

   private final ComboViewer comboViewer;

   /**
    *
    * Default constructor. Builds a TypeComposite.
    *
    * @param parent
    *           parent shell
    * @param style
    *           the style
    */
   public AircraftSelectionComposite(Composite parent) {
      super(parent, SWT.NONE);
      setLayout(new FormLayout());
      BundleContext bundleContext = FrameworkUtil.getBundle(IAircraftBusiness.class).getBundleContext();
      IAircraftBusiness aircraftBusiness = bundleContext.getService(bundleContext.getServiceReference(IAircraftBusiness.class));
      Label label = new Label(this, SWT.NONE);
      FormData fdLabel = new FormData();
      fdLabel.top = new FormAttachment(0, 6);
      label.setLayoutData(fdLabel);
      label.setText(Messages.CORE.getMnemonicEscapedString("FlightSynthesis_preference_Aircraft"));

      /**
       * Adding the type of Aircraft type
       */
      comboViewer = new ComboViewer(this, SWT.READ_ONLY);
      comboViewer.setContentProvider(ArrayContentProvider.getInstance());
      comboViewer.setLabelProvider(new LabelProvider() {
         @Override
         public String getText(Object arg0) {
            String stg = null;
            if (arg0 instanceof Aircraft model) {
               stg = model.getRegisterNumber();
            }
            return stg;
         }
      });

      List<Aircraft> activeAircrafts = aircraftBusiness.getAircrafts(AircraftState.ACTIVE);
      List<Aircraft> aircraftsWithHealthConf = new ArrayList<>();
      AircraftConfiguration aircraftConf;
      for (Aircraft aircraft : activeAircrafts) {
         try {
            aircraftConf = aircraftBusiness.getAircraftConfiguration(aircraft.getId());
         } catch (EntityNotFoundException e) {
            LOGGER.warn("Entity not found", e);
            continue;
         }
         if (aircraftConf.getConfigurations().get(HealthMonitoringConfiguration.TYPE) != null) {
            aircraftsWithHealthConf.add(aircraft);
         }
      }

      comboViewer.setInput(aircraftsWithHealthConf);
      Combo combo = comboViewer.getCombo();
      fdLabel.right = new FormAttachment(100, -425);
      FormData fdTypeCombo = new FormData();
      fdTypeCombo.top = new FormAttachment(label, -3, SWT.TOP);
      fdTypeCombo.left = new FormAttachment(label, 12);
      combo.setLayoutData(fdTypeCombo);
      combo.setBackground(Display.getDefault().getSystemColor(SWT.COLOR_WHITE));

      // Information label for deviation creation process  Ranjit
      Label deviationInfoLabel = new Label(this, SWT.WRAP);
      FormData fdDeviationInfoLabel = new FormData();
      fdDeviationInfoLabel.top = new FormAttachment(combo, 10);
      fdDeviationInfoLabel.left = new FormAttachment(label, 0, SWT.LEFT);
      fdDeviationInfoLabel.right = new FormAttachment(100, -5);
      deviationInfoLabel.setLayoutData(fdDeviationInfoLabel);
      deviationInfoLabel.setText(Messages.CORE.getMnemonicEscapedString("DeviationManagementMessage"));
   }

   public ComboViewer getComboViewer() {
      return comboViewer;
   }

   public Aircraft getSelectedAircraft() {
      return (Aircraft) ((IStructuredSelection) comboViewer.getSelection()).getFirstElement();
   }
}
