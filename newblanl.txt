package com.eurocopter.mgs.business.health.model;

import com.eurocopter.mgs.datamodel.health.monitoring.Indicator;
import com.eurocopter.mgs.datamodel.health.monitoring.IndicatorDisplay;
import com.eurocopter.mgs.xml.configuration.unit.Unit;

public class IndicatorModel {

   private final Indicator indicator;
   private final IndicatorDisplay indicatorDisplay;
   private final Object[] availableUnits;
   private final WCThresholdModel[] thresholdModels;
   private String name;

   public IndicatorModel(Indicator indicator, IndicatorDisplay indicatorDisplay, Object[] availableUnits, WCThresholdModel[] thresholdModels) {
      this.indicator = indicator;
      this.indicatorDisplay = indicatorDisplay;
      this.availableUnits = availableUnits;
      this.thresholdModels = thresholdModels;
      this.name = indicator.getName();
   }

   public String getName() {
      return name;
   }

   public void setName(String name) {
      this.name = name;
   }

   public Indicator getIndicator() {
      return indicator;
   }

   public IndicatorDisplay getIndicatorDisplay() {
      return indicatorDisplay;
   }

   public Object[] getAvailableUnits() {
      return availableUnits;
   }

   public WCThresholdModel[] getThresholdModels() {
      return thresholdModels;
   }

   public String getUnitSymbol(String unitName) {
      for (Object o : getAvailableUnits()) {
         if (o instanceof Unit unit) {
            if (unit.getName().equalsIgnoreCase(unitName)) {
               return unit.getSymbol();
            }
         }
      }
      return null;
   }
}
