

import java.io.Serializable;
import java.util.HashSet;
import java.util.Set;

public class IndicatorConfigurationDto implements Serializable {

   private static final long serialVersionUID = 1L;

   private final String indicatorId;

   private final Set<ThresholdConfigurationDto> thresholdConfigurationDtos = new HashSet<>();

   public IndicatorConfigurationDto(String indicatorId) {
      this.indicatorId = indicatorId;
   }

   public final String getIndicatorId() {
      return indicatorId;
   }

   public final Set<ThresholdConfigurationDto> getThresholdConfigurationDtos() {
      return thresholdConfigurationDtos;
   }
}
