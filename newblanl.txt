

import java.io.Serializable;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

import com.eurocopter.mgs.datamodel.health.common.Accelerometer;
import com.eurocopter.mgs.datamodel.health.common.HealthAlgorithm;

public class IndicatorConfiguration implements Serializable {

   private static final long serialVersionUID = 1L;

   private String id;

   private Indicator indicator;

   private String reference;

   private String referenceIndicator;

   private Integer slidingMedian;

   private String healthAlgorithm;

   /**
    * Algorithm parameters.
    */
   private List<Double> parameters = new ArrayList<>();

   private Set<ThresholdConfiguration> thresholdConfigurations = new HashSet<>();

   /**
    * Accelerometer that monitors the component.
    */
   private Accelerometer accelerometer;

   /**
    * Filtering Law
    */
   private String filteringLaw;

   /**
    * Max points per hour
    */
   private Integer maxPointsPerHour;

   /**
    * Min Sensor anomaly
    */
   private Double minSensor;

   /**
    * Max Sensor Anomaly
    */
   private Double maxSensor;

   private String sensorAnomalyWorkcard;

   private Integer sensorAnomalySuccessivePoints;

   private String invalidityWorkcard;

   private Double invalidityThreshold;

   private Double multiplier;

   IndicatorConfiguration() {
   }

   public IndicatorConfiguration(String id) {
      this.id = id;
   }

   public final String getId() {
      return id;
   }

   public final Indicator getIndicator() {
      return indicator;
   }

   public String getReference() {
      return reference;
   }

   public void setReference(String reference) {
      this.reference = reference;
   }

   public String getReferenceIndicator() {
      return referenceIndicator;
   }

   public void setReferenceIndicator(String referenceIndicator) {
      this.referenceIndicator = referenceIndicator;
   }

   public Integer getSlidingMedian() {
      return slidingMedian;
   }

   public void setSlidingMedian(Integer slidingMedian) {
      this.slidingMedian = slidingMedian;
   }

   public final void setIndicator(Indicator indicator) {
      this.indicator = indicator;
   }

   public final HealthAlgorithm getHealthAlgorithm() {
      return healthAlgorithm != null ? HealthAlgorithm.valueOf(healthAlgorithm) : null;
   }

   public final void setHealthAlgorithm(HealthAlgorithm healthAlgorithm) {
      this.healthAlgorithm = healthAlgorithm != null ? healthAlgorithm.name() : null;
   }

   public List<Double> getParameters() {
      return parameters;
   }

   public void setParameters(List<Double> parameters) {
      this.parameters = parameters;
   }

   public Set<ThresholdConfiguration> getThresholdConfigurations() {
      return thresholdConfigurations;
   }

   public void setThresholdConfigurations(Set<ThresholdConfiguration> thresholdConfigurations) {
      this.thresholdConfigurations = thresholdConfigurations;
   }

   public Accelerometer getAccelerometer() {
      return accelerometer;
   }

   public void setAccelerometer(Accelerometer accelerometer) {
      this.accelerometer = accelerometer;
   }

   public String getFilteringLaw() {
      return filteringLaw;
   }

   public void setFilteringLaw(String filteringLaw) {
      this.filteringLaw = filteringLaw;
   }

   public Integer getMaxPointsPerHour() {
      return maxPointsPerHour;
   }

   public void setMaxPointsPerHour(Integer maxPointsPerHour) {
      this.maxPointsPerHour = maxPointsPerHour;
   }

   public Double getMinSensor() {
      return minSensor;
   }

   public void setMinSensor(Double minSensor) {
      this.minSensor = minSensor;
   }

   public Double getMaxSensor() {
      return maxSensor;
   }

   public void setMaxSensor(Double maxSensor) {
      this.maxSensor = maxSensor;
   }

   public String getSensorAnomalyWorkcard() {
      return sensorAnomalyWorkcard;
   }

   public void setSensorAnomalyWorkcard(String sensorAnomalyWorkcard) {
      this.sensorAnomalyWorkcard = sensorAnomalyWorkcard;
   }

   public Integer getSensorAnomalySuccessivePoints() {
      return sensorAnomalySuccessivePoints;
   }

   public void setSensorAnomalySuccessivePoints(Integer sensorAnomalySuccessivePoints) {
      this.sensorAnomalySuccessivePoints = sensorAnomalySuccessivePoints;
   }

   public Double getInvalidityThreshold() {
      return invalidityThreshold;
   }

   public void setInvalidityThreshold(Double invalidityThreshold) {
      this.invalidityThreshold = invalidityThreshold;
   }

   public String getInvalidityWorkcard() {
      return invalidityWorkcard;
   }

   public void setInvalidityWorkcard(String invalidityWorkcard) {
      this.invalidityWorkcard = invalidityWorkcard;
   }

   public Double getMultiplier() {
      return multiplier;
   }

   public void setMultiplier(Double multiplier) {
      this.multiplier = multiplier;
   }
}
