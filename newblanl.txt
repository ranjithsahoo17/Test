

import java.io.Serializable;

import jakarta.persistence.Column;
import jakarta.persistence.DiscriminatorColumn;
import jakarta.persistence.DiscriminatorType;
import jakarta.persistence.Entity;
import jakarta.persistence.FetchType;
import jakarta.persistence.Id;
import jakarta.persistence.Inheritance;
import jakarta.persistence.InheritanceType;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.ManyToOne;
import jakarta.persistence.MappedSuperclass;
import jakarta.persistence.NamedQuery;
import jakarta.persistence.Table;

@Entity
@Table(name = "THRESHOLD_CONFIGURATION")
@MappedSuperclass
@Inheritance(strategy = InheritanceType.SINGLE_TABLE)
@DiscriminatorColumn(name = "THRESHOLD_TYPE", discriminatorType = DiscriminatorType.STRING)
@NamedQuery(name = "findThresholdConfigurationByIndicator", query = "SELECT tc FROM HealthMonitoringConfiguration hmc JOIN hmc.mechanicalComponentConfigurations mcc JOIN mcc.indicatorConfigurations ic JOIN ic.thresholdConfigurations tc WHERE ic.indicator.id = :indicatorId AND hmc.id = :configurationId")
@NamedQuery(name = "findThresholdConfigurationsByThresholdId", query = "SELECT tc FROM HealthMonitoringConfiguration hmc JOIN hmc.mechanicalComponentConfigurations mcc JOIN mcc.indicatorConfigurations ic JOIN ic.thresholdConfigurations tc WHERE tc.threshold.id = :thresholdId")
public abstract class ThresholdConfiguration implements Serializable {

   private static final long serialVersionUID = 1L;

   @Id
   @Column(name = "ID", nullable = false)
   private String id;

   @ManyToOne(fetch = FetchType.EAGER, optional = false)
   @JoinColumn(name = "THRESHOLD_ID", nullable = false)
   private Threshold threshold;

   /**
    * The Level of the Threshold
    */
   @Column(name = "THRESHOLD_LEVEL")
   private String thresholdLevel;

   @Column(name = "NB_SUCCESIVE_POINTS")
   private Integer nbSuccessivePoints;

   /**
    * Minimum engine oil temperature.
    */
   @Column(name = "MINIMUM_ENGINE_OIL_TEMPERATURE")
   private Double minimumEngineOilTemperature;

   /**
    * Reference of the work card to apply when an alarm is raised.
    */
   @Column(name = "WORK_CARD")
   private String workcard;

   ThresholdConfiguration() {
   }

   public ThresholdConfiguration(String id) {
      this.id = id;
   }

   public final String getId() {
      return id;
   }

   public ThresholdLevel getThresholdLevel() {
      return thresholdLevel != null ? ThresholdLevel.valueOf(thresholdLevel) : null;
   }

   public void setThresholdLevel(ThresholdLevel thresholdLevel) {
      this.thresholdLevel = thresholdLevel != null ? thresholdLevel.name() : null;
   }

   public Threshold getThreshold() {
      return threshold;
   }

   public void setThreshold(Threshold threshold) {
      this.threshold = threshold;
   }

   public Integer getNbSuccessivePoints() {
      return nbSuccessivePoints;
   }

   public void setNbSuccessivePoints(Integer nbSuccessivePoints) {
      this.nbSuccessivePoints = nbSuccessivePoints;
   }

   public Double getMinimumEngineOilTemperature() {
      return minimumEngineOilTemperature;
   }

   public void setMinimumEngineOilTemperature(Double minimumEngineOilTemperature) {
      this.minimumEngineOilTemperature = minimumEngineOilTemperature;
   }

   public abstract Double getValue();

   /**
    * Gets the workcard.
    * 
    * @return the workcard
    */
   public String getWorkcard() {
      return workcard;
   }

   /**
    * Sets the workcard.
    * 
    * @param workcard
    *           the new workcard
    */
   public void setWorkcard(String workcard) {
      this.workcard = workcard;
   }

   @Override
   public int hashCode() {
      final int prime = 31;
      int result = 1;
      result = prime * result + ((id == null) ? 0 : id.hashCode());
      return result;
   }

   @Override
   public boolean equals(Object obj) {
      if (this == obj) {
         return true;
      }
      if (obj == null) {
         return false;
      }
      if (getClass() != obj.getClass()) {
         return false;
      }
      ThresholdConfiguration other = (ThresholdConfiguration) obj;
      if (id == null) {
         return other.id == null;
      } else
         return id.equals(other.id);
   }
}
