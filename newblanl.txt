package com.eurocopter.mgs.business.health.model;

import com.eurocopter.mgs.datamodel.alarm.Severity;
import com.eurocopter.mgs.datamodel.health.monitoring.Indicator;
import com.eurocopter.mgs.datamodel.health.monitoring.ThresholdLevel;
import com.eurocopter.mgs.datamodel.health.monitoring.ThresholdType;

public class WCThresholdModel {

   /** Used to alternate line colors. */
   private boolean odd;

   /** Indicator. */
   private final Indicator indicator;

   private final String healthMechanicalComponentName;

   /** Indicator severity (RED or AMBER). */
   private Severity severity;

   /** Indicate whether this indicator is enabled or not. */
   private Boolean enabled;

   /** Indicator threshold level (HIGH or LOW). */
   private ThresholdLevel level;

   /** Threshold type (LEARNED or FIXED). */
   private ThresholdType type;

   /** Threshold default value. */
   private Double defaultValue;

   /** Learned Threshold min value. */
   private Double minValue;

   /** Learned Threshold max value. */
   private Double maxValue;

   private Integer nbSuccessivePoints;

   /** Threshold learning time. */
   private Long learningTime;

   /** Work card reference. */
   private String workCardReference;

   /**
    * Mean factor for relearning value computation.
    */
   private Double meanFactor;

   /**
    * Deviation coefficient for relearning value computation.
    */
   private Double deviationCoefficient;

   /**
    * Minimum deviation for relearning value computation.
    */
   private Double minimumDeviation;

   /**
    * Minimum engine oil temperature.
    */
   private Double minimumEngineOilTemperature;

   /**
    * Constructor with argument. Builds a WCIndicatorModel object with default values.
    *
    * @param healthMonitoringConfiguration
    *
    * @param indicator
    *           the indicator
    */
   public WCThresholdModel(Indicator indicator, String healthMechanicalComponentName) {
      this.indicator = indicator;
      this.severity = null;
      this.enabled = Boolean.FALSE;
      this.level = ThresholdLevel.LOW;
      this.type = ThresholdType.FIXED;
      this.maxValue = 0.0d;
      this.learningTime = 0L;
      this.workCardReference = " ";
      this.healthMechanicalComponentName = healthMechanicalComponentName;
   }

   public boolean isOdd() {
      return odd;
   }

   /**
    * Sets the lineColor.
    *
    * @param odd
    *           indicates whether the line number is odd or not.
    */
   public void setOdd(boolean odd) {
      this.odd = odd;
   }

   public Indicator getIndicator() {
      return indicator;
   }

   public Severity getSeverity() {
      return severity;
   }

   /**
    * Sets the severity.
    *
    * @param severities
    *           the severity to set.
    */
   public void setSeverity(Severity severities) {
      this.severity = severities;
   }

   public Boolean isEnabled() {
      return enabled;
   }

   public void setEnabled(Boolean enabled) {
      this.enabled = enabled;
   }

   public ThresholdLevel getLevel() {
      return level;
   }

   public void setLevel(ThresholdLevel level) {
      this.level = level;
   }

   public ThresholdType getType() {
      return type;
   }

   public void setType(ThresholdType type) {
      this.type = type;
   }

   public Double getDefaultValue() {
      return defaultValue;
   }

   public void setDefaultValue(Double defaultValue) {
      this.defaultValue = defaultValue;
   }

   public Double getMinValue() {
      return minValue;
   }

   public void setMinValue(Double minValue) {
      this.minValue = minValue;
   }

   public Double getMaxValue() {
      return maxValue;
   }

   public void setMaxValue(Double maxValue) {
      this.maxValue = maxValue;
   }

   public Long getLearningTime() {
      return learningTime;
   }

   public void setLearningTime(Long learningTime) {
      this.learningTime = learningTime;
   }

   public String getWorkCardReference() {
      if (workCardReference == null || "null".equals(workCardReference)) {
         workCardReference = "";
      }
      return workCardReference;
   }

   public void setWorkCardReference(String workCardReference) {
      this.workCardReference = workCardReference;
   }

   @Override
   public String toString() {
      return indicator.getName();
   }

   public Double getMeanFactor() {
      return meanFactor;
   }

   public void setMeanFactor(Double meanFactor) {
      this.meanFactor = meanFactor;
   }

   public Double getDeviationCoefficient() {
      return deviationCoefficient;
   }

   public void setDeviationCoefficient(Double deviationCoefficient) {
      this.deviationCoefficient = deviationCoefficient;
   }

   public Double getMinimumDeviation() {
      return minimumDeviation;
   }

   public void setMinimumDeviation(Double minimumDeviation) {
      this.minimumDeviation = minimumDeviation;
   }

   public Integer getNbSuccessivePoints() {
      return nbSuccessivePoints;
   }

   public void setNbSuccessivePoints(Integer nbSuccessivePoint) {
      this.nbSuccessivePoints = nbSuccessivePoint;
   }

   public String getHealthMechanicalComponentName() {
      return healthMechanicalComponentName;
   }

   public Double getMinimumEngineOilTemperature() {
      return minimumEngineOilTemperature;
   }

   public void setMinimumEngineOilTemperature(Double minimumEngineOilTemperature) {
      this.minimumEngineOilTemperature = minimumEngineOilTemperature;
   }

}
