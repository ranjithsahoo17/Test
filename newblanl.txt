

import java.io.Serializable;

import com.eurocopter.mgs.datamodel.alarm.Severity;
import com.eurocopter.mgs.datamodel.health.monitoring.ThresholdLevel;
import com.eurocopter.mgs.datamodel.health.monitoring.ThresholdType;

public class ThresholdConfigurationDto implements Serializable {

   private static final long serialVersionUID = 1L;

   private final String thresholdSeverity;

   private final String thresholdLevel;

   private final String thresholdType;

   private Double value;

   private Long learningTime;

   private Double minimumThreshold;

   private Double maximumThreshold;

   private Double defaultValue;

   private Double meanFactor = 1.0;

   private Double deviationCoefficient;

   private Double minimumDeviation;

   private final Integer nbSuccessivePoints;

   private String workcard;

   private Double minimumEngineOilTemperature;

   private ThresholdConfigurationDto(Severity thresholdSeverity, ThresholdLevel thresholdLevel, ThresholdType thresholdType, Integer nbSuccessivePoints, Double minimumEngineOilTemperature) {
      this.thresholdSeverity = thresholdSeverity.name();
      this.thresholdLevel = thresholdLevel.name();
      this.thresholdType = thresholdType.name();
      this.nbSuccessivePoints = nbSuccessivePoints;
      this.minimumEngineOilTemperature = minimumEngineOilTemperature;
   }

   public ThresholdConfigurationDto(Severity thresholdSeverity, ThresholdLevel thresholdLevel, ThresholdType thresholdType, Double value, Integer nbSuccessivePoints,
         Double minimumEngineOilTemperature) {
      this(thresholdSeverity, thresholdLevel, thresholdType, nbSuccessivePoints, minimumEngineOilTemperature);
      this.value = value;
   }

   public ThresholdConfigurationDto(Severity thresholdSeverity, ThresholdLevel thresholdLevel, ThresholdType thresholdType, Long learningTime, Double minimumThreshold, Double maximumThreshold,
         Double defaultValue,
         Double minimumDeviation, Double meanFactor, Double deviationCoefficient, Integer nbSuccessivePoints, Double minimumEngineOilTemperature) {
      this(thresholdSeverity, thresholdLevel, thresholdType, nbSuccessivePoints, minimumEngineOilTemperature);
      this.learningTime = learningTime;
      this.minimumThreshold = minimumThreshold;
      this.maximumThreshold = maximumThreshold;
      this.defaultValue = defaultValue;
      this.value = defaultValue;
      this.minimumDeviation = minimumDeviation;
      this.meanFactor = meanFactor;
      this.deviationCoefficient = deviationCoefficient;
   }

   public final Severity getThresholdSeverity() {
      return Severity.valueOf(thresholdSeverity);
   }

   public final ThresholdLevel getThresholdLevel() {
      return ThresholdLevel.valueOf(thresholdLevel);
   }

   public final ThresholdType getThresholdType() {
      return ThresholdType.valueOf(thresholdType);
   }

   public final Double getValue() {
      return value;
   }

   public final Long getLearningTime() {
      return learningTime;
   }

   public Double getMinimumThreshold() {
      return minimumThreshold;
   }

   public final Double getMaximumThreshold() {
      return maximumThreshold;
   }

   public final Double getDefaultValue() {
      return defaultValue;
   }

   public final String getWorkcard() {
      return workcard;
   }

   public final void setWorkcard(String workcard) {
      this.workcard = workcard;
   }

   public Double getMeanFactor() {
      return meanFactor;
   }

   public Double getMinimumDeviation() {
      return minimumDeviation;
   }

   public Double getDeviationCoefficient() {
      return deviationCoefficient;
   }

   public Integer getNbSuccessivePoints() {
      return nbSuccessivePoints;
   }

   public Double getMinimumEngineOilTemperature() {
      return minimumEngineOilTemperature;
   }

}
